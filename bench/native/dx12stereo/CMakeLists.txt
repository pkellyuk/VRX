cmake_minimum_required(VERSION 3.24)
project(vrx_dx12stereo LANGUAGES CXX)
if(NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
    message(FATAL_ERROR "Native stereo requires 64-bit Windows")
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_executable(dx12_publisher_test publisher_test.cpp)
add_executable(d3d11_launch_fixture d3d11_fixture.cpp)
target_compile_definitions(d3d11_launch_fixture PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX)
target_link_libraries(d3d11_launch_fixture PRIVATE d3d11 dxgi d3dcompiler user32)
set_property(TARGET d3d11_launch_fixture PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
if(MSVC)
    target_compile_options(d3d11_launch_fixture PRIVATE /W4 /WX /EHsc)
endif()
add_executable(shared_gpu_transport_test ../xgstereo/gpu_transport_test.cpp)
add_executable(shared_pair_transport_test ../xgstereo/transport_test.cpp)
foreach(target dx12_publisher_test shared_gpu_transport_test shared_pair_transport_test)
    target_compile_definitions(${target} PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX)
    target_link_libraries(${target} PRIVATE d3d12 dxgi dxguid)
    if(MSVC)
        target_compile_options(${target} PRIVATE /W4 /WX /EHsc)
    endif()
endforeach()
target_link_libraries(dx12_publisher_test PRIVATE d3dcompiler)
enable_testing()
add_test(NAME native_dx12_same_frame COMMAND dx12_publisher_test)
add_test(NAME native_dx12_full_resolution COMMAND dx12_publisher_test --full-resolution)
add_test(NAME native_dx12_odd_extent COMMAND dx12_publisher_test --odd-extent)
add_test(NAME shared_gpu_leases COMMAND shared_gpu_transport_test)
add_test(NAME shared_pair_atomicity COMMAND shared_pair_transport_test)
add_test(NAME ordinary_d3d11_fixture COMMAND d3d11_launch_fixture --self-test)
set_tests_properties(ordinary_d3d11_fixture PROPERTIES TIMEOUT 30)
set_tests_properties(native_dx12_same_frame native_dx12_full_resolution native_dx12_odd_extent shared_gpu_leases shared_pair_atomicity PROPERTIES TIMEOUT 30)
